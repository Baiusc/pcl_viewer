cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(lidar_network C)
endif()
add_library(lidar_network STATIC network_config.c network_support.c network_address.c network_discovery.c device_identity.c)
target_compile_features(lidar_network PUBLIC c_std_11)
target_compile_options(lidar_network PRIVATE -Wall -Wextra -Werror)
target_include_directories(lidar_network PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(lidar_network PUBLIC m)

foreach(program lidar_netd lidar_netctl)
    add_executable(${program} ${program}.c ../third_party/cJSON/cJSON.c)
    target_link_libraries(${program} PRIVATE lidar_network)
    target_compile_options(${program} PRIVATE -Wall -Wextra -Werror)
    install(TARGETS ${program} RUNTIME DESTINATION libexec/lidar)
endforeach()

# 出厂写码工具只提供显式构建目标，不进入常规固件安装包。
add_executable(lidar_sn_provision EXCLUDE_FROM_ALL lidar_sn_provision.c)
target_compile_features(lidar_sn_provision PRIVATE c_std_11)
target_compile_options(lidar_sn_provision PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lidar_sn_provision PRIVATE lidar_network)
foreach(program lidar_network lidar_network_status)
    add_executable(${program}.cgi lidar_network_http.c ../third_party/cJSON/cJSON.c)
    target_link_libraries(${program}.cgi PRIVATE lidar_network)
    target_compile_options(${program}.cgi PRIVATE -Wall -Wextra -Werror)
endforeach()
target_compile_definitions(lidar_network_status.cgi PRIVATE LIDAR_NETWORK_DIAGNOSTIC=1)
install(TARGETS lidar_network.cgi RUNTIME DESTINATION www/cgi-bin)
install(TARGETS lidar_network_status.cgi RUNTIME DESTINATION libexec/lidar)
