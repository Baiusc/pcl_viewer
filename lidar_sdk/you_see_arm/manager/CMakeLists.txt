cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(lidar_management C)
endif()
if(NOT TARGET lidar_network)
    add_subdirectory(../network network)
endif()
add_executable(lidar_mgrd lidar_mgrd.c mgr_support.c ../core/ys_user_settings.c ../third_party/cJSON/cJSON.c
    ../third_party/iniparser/iniparser.c ../third_party/iniparser/dictionary.c)
target_compile_features(lidar_mgrd PRIVATE c_std_11)
target_compile_options(lidar_mgrd PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lidar_mgrd PRIVATE m lidar_network)
install(TARGETS lidar_mgrd RUNTIME DESTINATION bin)
add_executable(lidar_mgrctl lidar_mgrctl.c ../third_party/cJSON/cJSON.c)
target_compile_features(lidar_mgrctl PRIVATE c_std_11)
target_compile_options(lidar_mgrctl PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lidar_mgrctl PRIVATE m)
install(TARGETS lidar_mgrctl RUNTIME DESTINATION bin)

add_executable(lidar_crypto lidar_crypto.c)
target_compile_features(lidar_crypto PRIVATE c_std_11)
target_compile_options(lidar_crypto PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lidar_crypto PRIVATE dl)
install(TARGETS lidar_crypto RUNTIME DESTINATION bin)

add_executable(lidar_boot lidar_boot.c fpga_spi.c mgr_support.c ../core/ys_user_settings.c ../third_party/cJSON/cJSON.c
    ../third_party/iniparser/iniparser.c ../third_party/iniparser/dictionary.c)
target_compile_features(lidar_boot PRIVATE c_std_11)
target_compile_options(lidar_boot PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lidar_boot PRIVATE m)
install(TARGETS lidar_boot RUNTIME DESTINATION bin)
